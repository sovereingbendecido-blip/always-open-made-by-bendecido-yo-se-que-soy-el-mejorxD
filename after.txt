local function getQueueFunc()
    local candidates = {
        rawget(getfenv(), "queue_on_teleport"),
        rawget(getfenv(), "queueteleport"),
        syn and syn.queue_on_teleport,
        fluxus and fluxus.queue_on_teleport,
        krnl and krnl.queue_on_teleport,
    }
    for _, f in ipairs(candidates) do
        if type(f) == "function" then return f end
    end
    return nil
end

local QueueOnTeleport = getQueueFunc()

local function setupPersistence()
    if not QueueOnTeleport then
        warn("[afterparty] Executor sin queue_on_teleport. Persistencia no disponible.")
        return
    end

    local payload = string.format([[
        local ok, err = pcall(function()
            loadstring(game:HttpGet("%s"))()
        end)
        if not ok then warn("[afterparty] Error al recargar: " .. tostring(err)) end
    ]], SCRIPT_URL)

    local ok, err = pcall(QueueOnTeleport, payload)
    if ok then
        print("[afterparty] Persistencia activada. Se re-ejecutará al cambiar de servidor/juego.")
    else
        warn("[afterparty] Falló queue_on_teleport: " .. tostring(err))
    end

    pcall(function()
        game:GetService("Players").LocalPlayer.OnTeleport:Connect(function()
            pcall(QueueOnTeleport, payload)
        end)
    end)
end

local function bindToString(b)
    if not b then return "" end
    if b.type == "key" then return "key:" .. tostring(b.value.Name) end
    if b.type == "mouse" then return "mouse:" .. tostring(b.value.Name) end
    return ""
end

local function stringToBind(s)
    if not s or s == "" then return nil end
    local t, v = s:match("^(%a+):(.+)$")
    if not t or not v then return nil end
    if t == "key" then
        local k = Enum.KeyCode[v]
        if k then return { type = "key", value = k } end
    elseif t == "mouse" then
        local m = Enum.UserInputType[v]
        if m then return { type = "mouse", value = m } end
    end
    return nil
end

    local c = CodeTP
    local e = ESP.Settings

    if data.ctpEmulateM1 then c.EmulateM1 = data.ctpEmulateM1 == "true" end
    if data.ctpAutoSquad then c.AutoSquadTP = data.ctpAutoSquad == "true" end
    if data.espEnabled then ESP.Enabled = data.espEnabled == "true" end
    if data.espGlow then e.ESPGlow = data.espGlow == "true" end
    if data.espNumHP then e.ESPNumHP = data.espNumHP == "true" end
    if data.espShowDist then e.ShowDist = data.espShowDist == "true" end
    if data.espShowMD then e.ShowMD = data.espShowMD == "true" end
    if data.espShowFov then e.ShowFov = data.espShowFov == "true" end

    if data.espRadar then e.ShowRadar = data.espRadar == "true" end

local DP = { _active = false, _start = nil, _stop = nil, _singleShot = nil,
    _setStrength = nil, _getStrength = nil, _isRunning = nil,
    _setSingleShot = nil, _getSingleShot = nil, _setSelfAntiLag = nil,

    _token = 0, _running = false, _selfAntiLag = true, _strength = 3,
    _cleanup = nil }

local function DP_Enable()
    if DP._active then return end
    DP._active = true
    local alive = true
    local hookInstalled = false
    local blockConns = {}
    local strength = DP._strength

    local function safe(fn) local ok, r = pcall(fn); if ok then return r end end
    local function newToken() DP._token += 1; return DP._token end

    local function installHook()
        if hookInstalled then return end
        local remote = LP:FindFirstChild("startevent")
        if not remote then return end
        safe(function()
            local orig = remote.FireServer
            hookfunction(orig, function(self, ...)
                if DP._fireBlocked then return nil end
                return orig(self, ...)
            end)
            hookInstalled = true
        end)
    end

    local function nukeObj(obj)
        if not obj or not obj.Parent then return end
        safe(function()
            for _, d in ipairs(obj:GetDescendants()) do
                if d:IsA("LocalScript") or d:IsA("Script") then d.Disabled = true end
            end
            obj:Destroy()
        end)
    end

    local function shouldNuke(obj)
        local n = string.lower(obj.Name)
        return n == "airbendstaff" or n == "puppetart" or n == "windstaff" or n == "puppetcontrol"
    end

    local function stripToolTextures(tool)
        if not tool then return end
        safe(function()
            for _, d in ipairs(tool:GetDescendants()) do
                if d:IsA("Decal") or d:IsA("Texture") then pcall(function() d:Destroy() end)
                elseif d:IsA("SpecialMesh") then pcall(function() d.TextureId = "" end)
                elseif d:IsA("MeshPart") then pcall(function() d.TextureID = "" end)
                elseif d:IsA("ParticleEmitter") or d:IsA("Trail") or d:IsA("Beam") then pcall(function() d.Enabled = false; d:Destroy() end)
                elseif d:IsA("Sound") then pcall(function() d:Stop(); d.Volume = 0 end)
                elseif d:IsA("Attachment") then
                    if string.lower(d.Name) == "puppetcontrol" then pcall(function() d:Destroy() end) end
                end
            end
        end)
    end

    local function nukeAll()
        safe(function()
            local char = LP.Character
            if char then
                for _, c in ipairs(char:GetChildren()) do
                    if shouldNuke(c) then
                        if c:IsA("Tool") then stripToolTextures(c) else nukeObj(c) end
                    end
                end
                for _, d in ipairs(char:GetDescendants()) do
                    if d:IsA("Attachment") and string.lower(d.Name) == "puppetcontrol" then
                        safe(function() d:Destroy() end)
                    end
                end
            end
        end)
        safe(function()
            local pp = workspace:FindFirstChild("projectileparent")
            if not pp then return end
            local pf = pp:FindFirstChild(LP.Name .. "mouse")
            if not pf then return end
            for _, c in ipairs(pf:GetChildren()) do if shouldNuke(c) then nukeObj(c) end end
        end)
    end

    local _blockerRunning = false
    local function startBlocker()
        if _blockerRunning then return end
        _blockerRunning = true
        nukeAll()
        local function hookChar(char)
            if not char then return end
            table.insert(blockConns, char.ChildAdded:Connect(function(c)
                if shouldNuke(c) then
                    if c:IsA("Tool") then stripToolTextures(c) else nukeObj(c) end
                end
            end))
            table.insert(blockConns, char.DescendantAdded:Connect(function(d)
                if d:IsA("Attachment") and string.lower(d.Name) == "puppetcontrol" then
                    safe(function() d:Destroy() end)
                end
            end))
        end
        hookChar(LP.Character)
        table.insert(blockConns, LP.CharacterAdded:Connect(hookChar))
        safe(function()
            local pp = workspace:FindFirstChild("projectileparent")
            if not pp then return end
            local function watchFolder(pf)
                if not pf or pf.Name ~= LP.Name .. "mouse" then return end
                table.insert(blockConns, pf.ChildAdded:Connect(function(c) if shouldNuke(c) then nukeObj(c) end end))
            end
            watchFolder(pp:FindFirstChild(LP.Name .. "mouse"))
            table.insert(blockConns, pp.ChildAdded:Connect(watchFolder))
        end)
        table.insert(blockConns, RunService.Heartbeat:Connect(nukeAll))
    end

    local function stopBlocker()
        if not _blockerRunning then return end
        _blockerRunning = false
        for _, c in ipairs(blockConns) do safe(function() c:Disconnect() end) end
        blockConns = {}
    end

    local singleShotMode = false

    local function fireRemote(toggle, myToken)
        if DP._token ~= myToken then return end
        if DP._fireBlocked then return end
        local remote = LP:FindFirstChild("startevent")
        local folder = ReplicatedStorage:FindFirstChild("alljutsu")
        if folder then folder = folder:FindFirstChild("FIGHTINGSTYLES") end
        if folder then folder = folder:FindFirstChild(toggle and "puppetart" or "airbending") end
        if not remote or not folder then return end
        safe(function() remote:FireServer("selectmartialstyle", folder) end)
    end

    local function lagLoopP(myToken, batch, waitT)
        local toggle = math.random() > 0.5
        while alive and DP._token == myToken do
            for i = 1, batch do
                if DP._token ~= myToken then return end
                fireRemote(toggle, myToken); toggle = not toggle
            end
            task.wait(waitT)
        end
    end

    local function megaCleaner(myToken)
        while alive and DP._token == myToken do
            nukeAll(); task.wait(0.05)
        end
    end

    local function stopAll()
        DP._fireBlocked = true
        newToken()
        DP._running = false
        task.wait(0.5)
        nukeAll()
        task.delay(1, function() DP._fireBlocked = false end)
    end

    local function startRun()
        if DP._running then return end
        installHook()
        DP._fireBlocked = false
        DP._running = true
        local myToken = newToken()
        local lvl = strength
        local threads, batch, waitT
        if lvl <= 2 then
            threads = lvl; batch = lvl * 5; waitT = 0.16 - lvl * 0.015
        else
            threads = math.clamp(math.floor(lvl / 2), 1, 25)
            batch = lvl * 8
            waitT = (lvl >= 25) and 0.001 or math.max(0.001, 0.18 - lvl * 0.006)
        end
        for i = 1, threads do task.spawn(function() lagLoopP(myToken, batch, waitT) end) end
        task.spawn(function() megaCleaner(myToken) end)
        if lvl >= 25 then
            task.spawn(function()
                while alive and DP._token == myToken do
                    task.wait(0.3)
                    if DP._token ~= myToken then break end
                    task.spawn(function() lagLoopP(myToken, 100, 0.001) end)
                    task.spawn(function() lagLoopP(myToken, 100, 0.001) end)
                    task.spawn(function() lagLoopP(myToken, 100, 0.001) end)
                end
            end)
        end
    end

    local function singleShot()
        if DP._running then return end
        installHook()
        DP._fireBlocked = false
        DP._running = true
        local myToken = newToken()
        local lvl = strength
        local batch = lvl <= 2 and lvl * 5 or lvl * 8
        task.spawn(function()
            local toggle = math.random() > 0.5
            for i = 1, batch do
                if DP._token ~= myToken then return end
                fireRemote(toggle, myToken); toggle = not toggle
            end
            DP._fireBlocked = true
            newToken(); DP._running = false
            task.wait(0.3); nukeAll()
            task.delay(0.5, function() DP._fireBlocked = false end)
        end)
    end

    DP._start = startRun
    DP._stop = stopAll
    DP._singleShot = singleShot
    DP._startBlocker = startBlocker
    DP._stopBlocker = stopBlocker
    DP._setSingleShot = function(v) singleShotMode = v end
    DP._getSingleShot = function() return singleShotMode end
    DP._setSelfAntiLag = function(v) DP._selfAntiLag = v end
    DP._setStrength = function(n)
        n = tonumber(n)
        if n then strength = math.clamp(math.floor(n), 1, 50); DP._strength = strength end
        return strength
    end
    DP._getStrength = function() return strength end
    DP._isRunning = function() return DP._running end

    DP._cleanup = function()
        alive = false
        DP._active = false
        safe(stopAll)
        safe(stopBlocker)
        DP._start, DP._stop, DP._singleShot = nil, nil, nil
    end
end

DP.SetEnabled = function(on)
    if on then DP_Enable()
    else if DP._cleanup then pcall(DP._cleanup) end; DP._cleanup = nil; DP._active = false end
end
DP.IsRunning = function() return DP._isRunning and DP._isRunning() or false end
DP.Start = function() if DP._start then DP._start() end end

DP.SingleShot = function() if DP._singleShot then DP._singleShot() end end
DP.SetSingleShot = function(v) if DP._setSingleShot then DP._setSingleShot(v) end end

DP.GetStrength = function() return DP._getStrength and DP._getStrength() or 3 end

local function stopAllTracks() 
    for _, tr in ipairs(Flight.Tracks) do pcall(function() tr:Stop(0.15) end) end
    Flight.Tracks = {} 
end

local function playAnims(hum)
    stopAllTracks()
    local specs = { 
        { "rbxassetid://13197194894", Enum.AnimationPriority.Action4 }, 
        { "rbxassetid://8461421332", Enum.AnimationPriority.Action2 }, 
        { "rbxassetid://8461403091", Enum.AnimationPriority.Action } 
    }
    for _, s in ipairs(specs) do 
        pcall(function() 
            local a = Instance.new("Animation")
            a.AnimationId = s[1]
            local tr = hum:LoadAnimation(a)
            tr.Priority = s[2]; tr.Looped = true; tr:Play(0.15)
            table.insert(Flight.Tracks, tr)
        end) 
    end
end

local function stopFlight()
    if not Flight.Active then return end
    Flight.Active = false
    if Flight.HB then pcall(function() Flight.HB:Disconnect() end); Flight.HB = nil end
    stopAllTracks()
    if Flight.BV then pcall(function() Flight.BV:Destroy() end); Flight.BV = nil end
    if Flight.BG then pcall(function() Flight.BG:Destroy() end); Flight.BG = nil end
    local char = LP.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then hum.PlatformStand = false end
        local root = char:FindFirstChild("HumanoidRootPart")
        if root then 
            local a = root:FindFirstChild("CelestialBV"); if a then a:Destroy() end
            local b = root:FindFirstChild("CelestialBG"); if b then b:Destroy() end 
        end
    end
    pcall(function() TweenService:Create(workspace.CurrentCamera, TweenInfo.new(0.7, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), { FieldOfView = 70 }):Play() end)
end

local function startFlight()
    if Flight.Active or not CFG.Enabled or Freecam.Active then return end
    local char = LP.Character; if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end
    
    Flight.Active, Flight.StartTime = true, tick()
    local o1 = root:FindFirstChild("CelestialBV"); if o1 then o1:Destroy() end
    local o2 = root:FindFirstChild("CelestialBG"); if o2 then o2:Destroy() end
    
    local bv = Instance.new("BodyVelocity")
    bv.Name = "CelestialBV"; bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    bv.Velocity = Vector3.new(0, 0, 0); bv.P = 5000; bv.Parent = root
    
    local bg = Instance.new("BodyGyro")
    bg.Name = "CelestialBG"; bg.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    bg.P, bg.D = 5000, 500; bg.CFrame = root.CFrame; bg.Parent = root
    
    hum.PlatformStand = true
    Flight.BV, Flight.BG = bv, bg
    playAnims(hum)
    
    pcall(function() TweenService:Create(workspace.CurrentCamera, TweenInfo.new(0.45, Enum.EasingStyle.Sine, Enum.EasingDirection.In), { FieldOfView = 80 }):Play() end)
    
    if Flight.HB then pcall(function() Flight.HB:Disconnect() end) end
    Flight.HB = RunService.Heartbeat:Connect(function()
        if not Flight.Active then stopFlight(); return end
        local c = LP.Character; if not c then stopFlight(); return end
        local r = c:FindFirstChild("HumanoidRootPart"); if not r then stopFlight(); return end
        if CFG.Mode == "toggle" and CFG.Duration > 0 and (tick() - Flight.StartTime >= CFG.Duration) then stopFlight(); return end
        local cam = workspace.CurrentCamera
        local far = cam and (cam.CFrame * CFrame.new(0, 0, -6000))
        local dir = far and (far.Position - r.Position).Unit or (cam and cam.CFrame.LookVector or Vector3.new(0, 0, -1))
        Flight.BV.Velocity = dir * CFG.Speed
        Flight.BG.CFrame = CFrame.new(r.Position, r.Position + dir)
    end)
end

local function startFreecam()
    if Freecam.Active then return end
    local char = LP.Character; if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart"); if not root then return end
    if Flight.Active then stopFlight() end
    local cam = workspace.CurrentCamera
    Freecam.CamCFrame = cam.CFrame
    Freecam.OriginalCamMode = LP.CameraMode
    root.CFrame = root.CFrame + Vector3.new(0, Freecam.BodyOffset, 0)
    cam.CameraType = Enum.CameraType.Scriptable
    cam.CFrame = Freecam.CamCFrame
    Freecam.Active = true
    Freecam.CamConn = RunService.RenderStepped:Connect(function()
        if not Freecam.Active then return end
        local c = workspace.CurrentCamera; if not c then return end
        local shiftHeld = UIS:IsKeyDown(Enum.KeyCode.LeftShift) or UIS:IsKeyDown(Enum.KeyCode.RightShift)
        local speed = shiftHeld and 1.5 or 0.5
        Freecam.ArmHeld = shiftHeld
        local mv = Vector3.new()
        if UIS:IsKeyDown(Enum.KeyCode.W) then mv = mv + Vector3.new(0, 0, 1) end
        if UIS:IsKeyDown(Enum.KeyCode.S) then mv = mv + Vector3.new(0, 0, -1) end
        if UIS:IsKeyDown(Enum.KeyCode.A) then mv = mv + Vector3.new(-1, 0, 0) end
        if UIS:IsKeyDown(Enum.KeyCode.D) then mv = mv + Vector3.new(1, 0, 0) end
        if UIS:IsKeyDown(Enum.KeyCode.Space) then mv = mv + Vector3.new(0, 1, 0) end
        if UIS:IsKeyDown(Enum.KeyCode.Q) then mv = mv + Vector3.new(0, -1, 0) end
        local camCF = c.CFrame
        if mv.Magnitude > 0 then
            mv = mv.Unit * speed
            local newPos = camCF.Position + (camCF.LookVector * mv.Z) + (camCF.RightVector * mv.X) + (camCF.UpVector * mv.Y)
            c.CFrame = CFrame.new(newPos, newPos + camCF.LookVector)
        end
        if UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
            UIS.MouseBehavior = Enum.MouseBehavior.LockCurrentPosition
            local d = UIS:GetMouseDelta()
            local sens = 0.003
            c.CFrame = c.CFrame * CFrame.Angles(0, -d.X * sens, 0) * CFrame.Angles(-d.Y * sens, 0, 0)
        else UIS.MouseBehavior = Enum.MouseBehavior.Default end
    end)
end

    Freecam.Active = false
    Freecam.ArmHeld = false
    if Freecam.CamConn then Freecam.CamConn:Disconnect(); Freecam.CamConn = nil end
    UIS.MouseBehavior = Enum.MouseBehavior.Default
    local cam = workspace.CurrentCamera
    local char = LP.Character
    if char then
        local root = char:FindFirstChild("HumanoidRootPart")
        if root and cam then 
            root.CFrame = CFrame.new(cam.CFrame.Position)
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero

local function tpToTarget()
    if tick() - TP.Cooldown < TP.CooldownTime then return end
    TP.Cooldown = tick()
    local cam = workspace.CurrentCamera
    local char = LP.Character
    if not cam or not char then return end
    local root = char:FindFirstChild("HumanoidRootPart"); if not root then return end
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = { char, cam }
    params.IgnoreWater = true
    local targetPlayer = nil
    local result = workspace:Raycast(cam.CFrame.Position, cam.CFrame.LookVector * 50000, params)
    if result then
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LP and plr.Character and result.Instance:IsDescendantOf(plr.Character) then targetPlayer = plr; break end
        end
    end
    if not targetPlayer then
        local camLook = cam.CFrame.LookVector; local camPos = cam.CFrame.Position; local bestScore = math.huge
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LP and plr.Character then
                local tr = plr.Character:FindFirstChild("HumanoidRootPart")
                local th = plr.Character:FindFirstChildOfClass("Humanoid")
                if tr and th and th.Health > 0 then
                    local toTarget = tr.Position - camPos; local dist = toTarget.Magnitude
                    if dist > 0.1 then
                        local dot = camLook:Dot(toTarget.Unit)
                        if dot > 0.85 then local score = (1 - dot) * 1000 + dist * 0.001; if score < bestScore then bestScore = score; targetPlayer = plr end end
                    end
                end
            end
        end
    end
    if not targetPlayer then return end
    local targetChar = targetPlayer.Character
    local targetRoot = targetChar and targetChar:FindFirstChild("HumanoidRootPart")
    if targetRoot then 
        root.CFrame = targetRoot.CFrame * CFrame.new(0, 0, 3)
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero 
    end
end

    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if hum then hum.Health = 0 end
end

    if CodeTP.Busy or not CodeTP.Enabled then return end

    local fill = new("Frame", { Size = UDim2.new(0, 0, 1, 0), BackgroundColor3 = C.Accent, BorderSizePixel = 0 }, track); radius(fill, 2)

    local dragging = false
    local function apply(v, fire) 
        local n = math.clamp(math.floor(v + 0.5), min, max)
        local a = (n - min) / (max - min)
        fill.Size = UDim2.new(a, 0, 1, 0)
        thumb.Position = UDim2.new(a, 0, 0.5, 0)
        val.Text = tostring(n) .. suffix
        if fire then onChange(n) end 
    end
    local function fromX(x) 
        local rel = (x - track.AbsolutePosition.X) / track.AbsoluteSize.X
        apply(min + (max - min) * math.clamp(rel, 0, 1), true) 
    end
    track.InputBegan:Connect(function(i) 
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = true; fromX(i.Position.X) end 
    end)
    UIS.InputChanged:Connect(function(i) 
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then fromX(i.Position.X) end 
    end)
    UIS.InputEnded:Connect(function(i) 
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = false end 
    end)
    apply(def, false); return row
end

    local function GetESP(player)
        if not espDrawings[player] then
            local char = player.Character
            local head = char and char:FindFirstChild("Head")
            if not head then return nil end
            local bb, hpFill, hpBg, hpNumTxt, cdStk, cdTxt, nmTxt, distTxt, mdTxt = createHpBillboard(head)
            espDrawings[player] = {
                Glow = Instance.new("Highlight"),
                BB = bb, HpFill = hpFill, HpBg = hpBg, HpNumTxt = hpNumTxt,
                CdStroke = cdStk, CdTxt = cdTxt, NmTxt = nmTxt, DistTxt = distTxt, MdTxt = mdTxt,
                LastDodge = 0, FakeMD = 10000,
            }
            local d = espDrawings[player]
            d.Glow.FillTransparency = 0.5; d.Glow.OutlineTransparency = 0
            d.Glow.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            TrackPlayerDodges(player, d)
        end
        return espDrawings[player]
    end

    local function ClearESP(player)
        if espDrawings[player] then
            for k, v in pairs(espDrawings[player]) do
                if type(v) == "userdata" then
                    if k == "CharConn" or k == "ChildConn" or k == "StatConn" then
                        pcall(function() v:Disconnect() end)
                    elseif k ~= "LastDodge" and k ~= "FakeMD" then
                        pcall(function()
                            if v:IsA("BillboardGui") then v.Enabled = false end
                            if v:IsA("Highlight") then v.Adornee = nil end
                            v:Destroy()
                        end)
                    end
                end
            end
            espDrawings[player] = nil
        end
    end

    local function ClearRadar(player)
        if radarDots[player] then
            pcall(function() radarDots[player].Dot:Destroy() end)
            radarDots[player] = nil
        end
    end

    Players.PlayerRemoving:Connect(function(p) ClearESP(p); ClearRadar(p) end)

        if UI and UI.RefreshESP then UI:RefreshESP() end

            for _, p in pairs(Players:GetPlayers()) do
                local e = espDrawings[p]
                if e then e.BB.Enabled = false; if e.Glow then e.Glow.Adornee = nil end end
                if radarDots[p] then radarDots[p].Dot.Visible = false end
            end
            fovFrame.Visible = false; radarFrame.Visible = false
        end
        saveConfig()
    end
    function ESP.SetEnabled(on)

    RunService.RenderStepped:Connect(function()
        if not ESP.Enabled then return end
        if not _G.HUB_INJECTED and _G.HUB_INJECTED ~= nil then return end

        if ESP.Settings.ShowRadar then radarFrame.Visible = true
        else radarFrame.Visible = false; for _, d in pairs(radarDots) do d.Dot.Visible = false end end

        if ESP.Settings.ShowFov then
            fovFrame.Visible = true
            local viewport = ESP_cachedCam.ViewportSize
            local center = Vector2.new(viewport.X / 2, viewport.Y / 2)
            local r = ESP.Settings.FovSize or 150
            fovFrame.Position = UDim2.new(0, center.X, 0, center.Y)
            fovFrame.Size = UDim2.new(0, r * 2, 0, r * 2)
        else
            fovFrame.Visible = false
        end

        for _, p in pairs(Players:GetPlayers()) do
            if p == LP then continue end
            if ESP_IsSquadMember(p) then
                ClearESP(p); ClearRadar(p); continue
            end

            local char = p.Character
            local root = char and char:FindFirstChild("HumanoidRootPart")
            local hum  = char and char:FindFirstChild("Humanoid")
            local head = char and char:FindFirstChild("Head")

            if ESP.Settings.ShowRadar and char and root and hum and hum.Health > 0 then
                local maxR = ESP.Settings.RadarRange or 250
                local myChar = LP.Character
                local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
                local camPos = ESP_cachedCam.CFrame.Position
                local refPos = (Freecam.Active and camPos) or (myRoot and myRoot.Position) or camPos
                if refPos then
                    local dist = (root.Position - refPos).Magnitude
                    if dist <= maxR then
                        if not radarDots[p] then
                            local dot, txt, hpF = createRadarDot()
                            radarDots[p] = {Dot = dot, Txt = txt, HpF = hpF}
                        end
                        local rd = radarDots[p]
                        rd.Dot.Visible = true
                        local rel = ESP_cachedCam.CFrame:PointToObjectSpace(root.Position)
                        local angle = math.atan2(rel.X, -rel.Z)
                        local rScale = dist / maxR
                        local offset = rScale * (radarFrame.AbsoluteSize.X / 2 - 8)
                        local rx = math.sin(angle) * offset
                        local ry = -math.cos(angle) * offset
                        rd.Dot.Position = UDim2.new(0.5, rx - 4, 0.5, ry - 4)
                        rd.Dot.BackgroundColor3 = Color3.fromRGB(255, 50, 50)
                        rd.Txt.Text = p.Name .. " [" .. math.floor(dist) .. "m]"
                        local hpR = math.clamp(hum.Health/math.max(hum.MaxHealth,1), 0, 1)
                        rd.HpF.Size = UDim2.new(hpR, 0, 1, 0)
                        if hpR > 0.6 then rd.HpF.BackgroundColor3 = Color3.fromRGB(50, 255, 50)
                        elseif hpR > 0.3 then rd.HpF.BackgroundColor3 = Color3.fromRGB(255, 255, 50)
                        else rd.HpF.BackgroundColor3 = Color3.fromRGB(255, 50, 50) end
                    else
                        if radarDots[p] then radarDots[p].Dot.Visible = false end
                    end
                else
                    if radarDots[p] then radarDots[p].Dot.Visible = false end
                end
            else
                if radarDots[p] then radarDots[p].Dot.Visible = false end
            end

            if not char or not root or not hum or not head or hum.Health <= 0 then
                local e = espDrawings[p]
                if e then e.BB.Enabled = false; if e.Glow then e.Glow.Adornee = nil end end
                continue
            end

            local esp = espDrawings[p]
            if esp and not esp.BB.Parent then ClearESP(p); esp = nil end
            if not esp then esp = GetESP(p) end
            if not esp then continue end

            if esp.BB.Adornee ~= head then esp.BB.Adornee = head; esp.BB.Parent = head end

            local _, onScreen = ESP_cachedCam:WorldToViewportPoint(root.Position)
            if not onScreen then
                esp.BB.Enabled = false
                if esp.Glow then esp.Glow.Adornee = nil end
                continue
            end

            esp.BB.Enabled = true
            esp.NmTxt.Text = p.Name
            local maxHp = math.max(hum.MaxHealth, 1)
            local hpR = math.clamp(hum.Health / maxHp, 0, 1)
            esp.HpFill.Size = UDim2.new(hpR, 0, 1, 0)

            local hpColor
            if hpR > 0.6 then
                local t = (hpR - 0.6) / 0.4
                hpColor = Color3.fromRGB(math.floor(255*(1-t)), 220, 0)
            elseif hpR > 0.3 then
                local t = (hpR - 0.3) / 0.3
                hpColor = Color3.fromRGB(255, math.floor(200*t + 60*(1-t)), 0)
            else
                local t = hpR / 0.3
                hpColor = Color3.fromRGB(255, math.floor(60*t), 0)
            end
            esp.HpFill.BackgroundColor3 = hpColor

            if ESP.Settings.ESPNumHP then
                if esp.HpBg then esp.HpBg.Visible = false end
                if esp.HpNumTxt then
                    esp.HpNumTxt.Visible = true
                    esp.HpNumTxt.TextColor3 = hpColor
                    esp.HpNumTxt.Text = string.format("HP: %d/%d", math.floor(hum.Health), math.floor(maxHp))
                end
            else
                if esp.HpBg then esp.HpBg.Visible = true end
                if esp.HpNumTxt then esp.HpNumTxt.Visible = false end
            end

            if ESP.Settings.ShowDist then
                local dist = math.floor((ESP_cachedCam.CFrame.Position - root.Position).Magnitude)
                esp.DistTxt.Text = "[" .. tostring(dist) .. "m]"
                esp.DistTxt.Visible = true
            else
                esp.DistTxt.Visible = false
            end

            if ESP.Settings.ShowMD then
                local realMd = nil
                if char:FindFirstChild("combat") then
                    local modeVal = char.combat:FindFirstChild("mode")
                    if modeVal and (modeVal:IsA("IntValue") or modeVal:IsA("IntConstrainedValue") or modeVal:IsA("NumberValue")) then
                        realMd = tonumber(modeVal.Value)
                    end
                end
                if not realMd then
                    local lastMode = p:FindFirstChild("lastmode")
                    if lastMode and tonumber(lastMode.Value) then realMd = tonumber(lastMode.Value) end
                end
                local mdValue = realMd or 10000
                esp.MdTxt.Visible = true
                if mdValue >= 1000 then esp.MdTxt.Text = string.format("%.2fk", mdValue / 1000):gsub("%.", ",")
                else esp.MdTxt.Text = tostring(math.floor(mdValue)) end
                if mdValue < 1 then esp.MdTxt.TextColor3 = Color3.fromRGB(255, 50, 50)
                else esp.MdTxt.TextColor3 = Color3.fromRGB(170, 0, 255) end
            else
                esp.MdTxt.Visible = false
            end

            esp.CdStroke.Parent.Visible = true
            if tick() < esp.LastDodge then
                esp.CdTxt.Text = tostring(math.ceil(esp.LastDodge - tick()))
            else
                esp.CdTxt.Text = ""
            end

            if ESP.Settings.ESPGlow then
                esp.Glow.Adornee = char
                if not esp.Glow.Parent then esp.Glow.Parent = gui end
                if not esp.LastGlowCheck or tick() - esp.LastGlowCheck > 0.15 then
                    esp.LastGlowCheck = tick()
                    local origin = ESP_cachedCam.CFrame.Position
                    local params = RaycastParams.new()
                    params.FilterDescendantsInstances = {ESP_cachedCam, LP.Character}
                    params.FilterType = Enum.RaycastFilterType.Exclude
                    local result = workspace:Raycast(origin, head.Position - origin, params)
                    esp.IsVisibleCache = not (result and result.Instance and not result.Instance:IsDescendantOf(char))
                end
                local highlightColor = esp.IsVisibleCache and Color3.fromRGB(50, 255, 50) or Color3.fromRGB(255, 50, 50)
                esp.Glow.OutlineColor = highlightColor
                esp.Glow.FillColor = highlightColor
                esp.Glow.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            else
                if esp.Glow then esp.Glow.Adornee = nil end
            end
        end
    end)
end

local lastCtrl, lastW = 0, 0
UIS.InputBegan:Connect(function(input, gpe)
    if listening then
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.MouseButton2 or input.UserInputType == Enum.UserInputType.MouseButton3 then 
            CFG.Bind = { type = "mouse", value = input.UserInputType }
        elseif input.KeyCode ~= Enum.KeyCode.Unknown then CFG.Bind = { type = "key", value = input.KeyCode } 
        else return end
        listening = false; bindBtn.Text = inputLabel(input); bindBtn.TextColor3 = C.Accent2; saveConfig()
        return
    end
    if ctListening then 
        if input.KeyCode ~= Enum.KeyCode.Unknown then 
            CodeTP.Bind = input.KeyCode.Name; ctBindBtn.Text = CodeTP.Bind; ctBindBtn.TextColor3 = C.Accent2
            ctListening = false; ctp_rebind(); saveConfig()
        end
        return 
    end
    if gpe then return end
    if input.KeyCode == Enum.KeyCode.F1 then tpToTarget(); return end
    if input.KeyCode == Enum.KeyCode.F2 then ESP.Toggle(); return end
    if input.KeyCode == Enum.KeyCode.F3 then respawnSelf(); return end
    if input.KeyCode == Enum.KeyCode.RightControl then 
        local now = tick()
        if (now - lastCtrl) <= 0.28 then lastCtrl = 0; UI:Toggle() else lastCtrl = now end
        return 
    end
    if input.KeyCode == Enum.KeyCode.LeftAlt or input.KeyCode == Enum.KeyCode.RightAlt then
        local now = tick()
        if (now - Freecam.LastTap) <= Freecam.DoubleTapTime then Freecam.LastTap = 0; if Freecam.Active then stopFreecam() else startFreecam() end
        else Freecam.LastTap = now end
        return
    end
    if not CFG.Enabled then return end
    if Freecam.Active then return end
    if matches(input, CFG.Bind) then 
        if CFG.Mode == "hold" then startFlight() else if Flight.Active then stopFlight() else startFlight() end end
        return 
    end
    if not Flight.Active then return end
    if input.KeyCode == Enum.KeyCode.X and CFG.Bind.value ~= Enum.KeyCode.X then stopFlight(); return end
    if input.KeyCode == Enum.KeyCode.Space then stopFlight(); return end
    if input.KeyCode == Enum.KeyCode.W then local now = tick(); if (now - lastW) <= 0.32 then stopFlight(); return end; lastW = now end
end)

UIS.InputEnded:Connect(function(input)
    if CFG.Mode == "hold" and matches(input, CFG.Bind) and Flight.Active then stopFlight() end
end)
